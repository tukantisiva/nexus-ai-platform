3. PostgreSQL Schema (Supabase Free Tier) 

-- Enable Vector Extension in Supabase (Included Free)
CREATE EXTENSION IF NOT EXISTS vector;

-- Table 1: Ingested Intelligence Items
CREATE TABLE intelligence_items (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    title VARCHAR(255) NOT NULL,
    source VARCHAR(100) NOT NULL, -- ArXiv, GitHub, NVD, RSS
    url TEXT UNIQUE NOT NULL,
    raw_content TEXT,
    summary_markdown TEXT,
    category VARCHAR(50), -- Architecture, Security, Benchmark, News
    tags TEXT[],
    embedding VECTOR(768), -- Vector dimension for gemini-text-embedding-004
    published_at TIMESTAMP WITH TIME ZONE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- Table 2: Reference Architecture Blueprints
CREATE TABLE blueprints (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    title VARCHAR(255) NOT NULL,
    slug VARCHAR(100) UNIQUE NOT NULL,
    description TEXT,
    mermaid_code TEXT NOT NULL,
    use_cases TEXT[],
    tradeoffs TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- Table 3: Client Assessment Submissions
CREATE TABLE client_assessments (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    company_name VARCHAR(255) NOT NULL,
    contact_email VARCHAR(255) NOT NULL,
    score INTEGER NOT NULL,
    assessment_data JSONB NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- Free Vector Search RPC Function
CREATE OR REPLACE FUNCTION match_intelligence (
  query_embedding VECTOR(768),
  match_threshold FLOAT,
  match_count INT
)
RETURNS TABLE (
  id UUID,
  title VARCHAR,
  source VARCHAR,
  url TEXT,
  summary_markdown TEXT,
  similarity FLOAT
)
LANGUAGE plpgsql
AS $$
BEGIN
  RETURN QUERY
  SELECT
    intelligence_items.id,
    intelligence_items.title,
    intelligence_items.source,
    intelligence_items.url,
    intelligence_items.summary_markdown,
    1 - (intelligence_items.embedding <=> query_embedding) AS similarity
  FROM intelligence_items
  WHERE 1 - (intelligence_items.embedding <=> query_embedding) > match_threshold
  ORDER BY similarity DESC
  LIMIT match_count;
END;
$$;

3. Free GitHub Actions Ingestion Workflow

Junior developer should place this configuration file in .github/workflows/daily_ingest.yml:

name: Daily AI Intelligence Ingestion Pipeline

on:
  schedule:
    - cron: '0 6 * * *' # Runs daily at 06:00 UTC
  workflow_dispatch: # Allows manual trigger

jobs:
  ingest:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout Repository
        uses: actions/checkout@v4

      - name: Set up Python
        uses: actions/setup-python@v5
        with:
          python-version: '3.11'

      - name: Install Dependencies
        run: |
          pip install feedparser requests supabase google-genai

      - name: Execute Ingestion Script
        env:
          SUPABASE_URL: ${{ secrets.SUPABASE_URL }}
          SUPABASE_SERVICE_ROLE_KEY: ${{ secrets.SUPABASE_SERVICE_ROLE_KEY }}
          GEMINI_API_KEY: ${{ secrets.GEMINI_API_KEY }}
        run: python scripts/daily_ingest.py
